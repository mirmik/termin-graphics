cmake_minimum_required(VERSION 3.19)
project(termin_graphics_sdk VERSION 0.1.0 LANGUAGES C CXX)

option(TERMIN_ENABLE_OPENGL "Enable the desktop OpenGL backend" ON)
option(TERMIN_ENABLE_VULKAN "Enable the Vulkan backend" ON)
option(TERMIN_ENABLE_SDL "Enable SDL window hosting" ON)
option(TERMIN_BUILD_PYTHON "Build Python/nanobind bindings" OFF)
option(TERMIN_BUILD_TESTS "Build Graphics native tests" OFF)
option(TERMIN_BUILD_TGFX2_TESTS "Build tgfx2 native tests" ON)
option(TERMIN_BUILD_WINDOW_TESTS "Build tests requiring a window system" OFF)
option(TERMIN_USE_CCACHE "Use ccache when available" ON)
option(TERMIN_ENABLE_UNITY_BUILD "Enable CMake unity builds" OFF)
option(TERMIN_ENABLE_PCH "Enable precompiled headers" ON)
set(TERMIN_CORE_SDK "" CACHE PATH "Absolute path to an installed Core SDK")
set(TERMIN_CORE_BUILD_ID "" CACHE STRING
    "Optional expected native_build_id of the installed Core SDK")

if(NOT IS_ABSOLUTE "${TERMIN_CORE_SDK}")
    message(FATAL_ERROR
        "termin-graphics requires -DTERMIN_CORE_SDK with an absolute installed Core SDK path")
endif()
set(_termin_core_manifest "${TERMIN_CORE_SDK}/termin-artifacts.json")
if(NOT EXISTS "${_termin_core_manifest}")
    message(FATAL_ERROR "Installed Core artifact manifest is missing: ${_termin_core_manifest}")
endif()
file(READ "${_termin_core_manifest}" _termin_core_json)
string(REGEX MATCH
    "\"native_build_id\"[ \t\r\n]*:[ \t\r\n]*\"([^\"]+)\""
    _termin_core_id_match "${_termin_core_json}")
set(_termin_core_manifest_id "${CMAKE_MATCH_1}")
if(NOT _termin_core_manifest_id)
    message(FATAL_ERROR "Installed Core manifest has no native_build_id: ${_termin_core_manifest}")
endif()
if(TERMIN_CORE_BUILD_ID AND
   NOT TERMIN_CORE_BUILD_ID STREQUAL _termin_core_manifest_id)
    message(FATAL_ERROR
        "Installed Core identity mismatch: expected ${TERMIN_CORE_BUILD_ID}, "
        "got ${_termin_core_manifest_id}")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

if(TERMIN_USE_CCACHE AND NOT MSVC)
    find_program(TERMIN_CCACHE_PROGRAM ccache)
    if(TERMIN_CCACHE_PROGRAM)
        set(CMAKE_C_COMPILER_LAUNCHER "${TERMIN_CCACHE_PROGRAM}" CACHE STRING "" FORCE)
        set(CMAKE_CXX_COMPILER_LAUNCHER "${TERMIN_CCACHE_PROGRAM}" CACHE STRING "" FORCE)
    else()
        message(STATUS "Termin Graphics: ccache requested but not found")
    endif()
endif()

function(termin_enable_unity_build_for target)
    if(TERMIN_ENABLE_UNITY_BUILD AND TARGET "${target}")
        set_target_properties("${target}" PROPERTIES
            UNITY_BUILD ON
            UNITY_BUILD_BATCH_SIZE 8)
    endif()
endfunction()

function(termin_enable_pch_for target)
    if(TERMIN_ENABLE_PCH AND TARGET "${target}")
        get_target_property(_termin_target_type "${target}" TYPE)
        if(NOT MSVC OR NOT _termin_target_type STREQUAL "SHARED_LIBRARY")
            target_precompile_headers("${target}" PRIVATE
                "$<$<COMPILE_LANGUAGE:CXX>:${CMAKE_CURRENT_FUNCTION_LIST_DIR}/cmake/termin_cpp_pch.hpp>")
        endif()
    endif()
endfunction()

    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/termin-thirdparty/guard/guard_c.h" OR
       NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/termin-thirdparty/guard/guard_main.h")
        message(FATAL_ERROR
            "TERMIN_BUILD_TESTS=ON requires termin-thirdparty/guard")
    endif()
    enable_testing()

    foreach(_termin_aggregate IN ITEMS
